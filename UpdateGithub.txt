

#!/bin/bash

# Script: update_github_actions_ips.sh
# Función: Descargar las IPs oficiales de GitHub Actions y aplicarlas a iptables
# Uso: Ejecutar como root

set -e

# Archivo temporal donde se guardan las IPs descargadas
TMPFILE="/tmp/github_actions_ips.txt"

echo "[INFO] Descargando IPs oficiales de GitHub Actions..."
curl -s https://api.github.com/meta | jq -r '.actions[]' > "$TMPFILE"

if [ ! -s "$TMPFILE" ]; then
    echo "[ERROR] No se pudieron obtener IPs de GitHub Actions."
    exit 1
fi

echo "[INFO] IPs descargadas correctamente:"
cat "$TMPFILE"

echo "[INFO] Limpiando reglas anteriores de GitHub Actions..."
# Marcamos nuestras reglas con comentarios
for ip in $(cat "$TMPFILE"); do
    # Esta eliminación solo borra reglas coincidentes, sin errores
    iptables -D INPUT -p tcp -s "$ip" --dport 22 -j ACCEPT 2>/dev/null || true
done

echo "[INFO] Volviendo a bloquear SSH (regla general, opcional si ya existe)"
iptables -D INPUT -p tcp --dport 22 -j DROP 2>/dev/null || true
iptables -A INPUT -p tcp --dport 22 -j DROP

echo "[INFO] Agregando nuevas reglas para permitir SSH solo desde IPs de GitHub Actions..."
for ip in $(cat "$TMPFILE"); do
    iptables -I INPUT -p tcp -s "$ip" --dport 22 -j ACCEPT
    echo "   + Permitido: $ip"
done

echo "[INFO] Guardando reglas..."
# Para Ubuntu (netfilter-persistent)
if command -v netfilter-persistent &> /dev/null; then
    netfilter-persistent save
elif command -v iptables-save &> /dev/null; then
    iptables-save > /etc/iptables/rules.v4
fi

echo "[OK] Script completado."
echo "[OK] Solo las IPs oficiales de GitHub Actions pueden acceder al puerto 22."
